CONFIGURAR alembic/env.py
Abre backend/alembic/env.py y reemplázalo por esto:
from logging.config import fileConfig

from alembic import context
from sqlalchemy import engine_from_config, pool

from app.core.config import settings
from app.models import Base

# Config de Alembic
config = context.config

# Configurar logging
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

# URL de la base de datos desde settings
config.set_main_option("sqlalchemy.url", settings.DATABASE_URL)

# Metadata de los modelos
target_metadata = Base.metadata


def run_migrations_offline() -> None:
    """Correr migraciones en modo offline."""
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Correr migraciones en modo online."""
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=target_metadata,
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

CREAR app/core/config.py
Crea el archivo backend/app/core/config.py:
from pydantic_settings import BaseSettings, SettingsConfigDict


class Settings(BaseSettings):
    """Configuración de la aplicación."""
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        case_sensitive=True,
    )

    # Base de datos
    DATABASE_URL: str = (
        "postgresql://boe_user:boe_password@localhost:5432/boe_dev"
    )

    # Redis
    REDIS_URL: str = "redis://localhost:6379/0"

    # JWT
    SECRET_KEY: str = "cambiar-en-produccion"
    ALGORITHM: str = "HS256"
    ACCESS_TOKEN_EXPIRE_MINUTES: int = 30

    # App
    APP_NAME: str = "BoE_Mpr"
    APP_VERSION: str = "0.1.0"
    DEBUG: bool = True


settings = Settings()

\q

1	alembic revision --autogenerate -m "mensaje"	Genera migración automática
2	alembic upgrade head	Aplica todas las migraciones
3	alembic downgrade -1	Revierte la última migración
4	alembic current	Muestra la migración actual
5	alembic history	Muestra el historial de migraciones
6	alembic heads	Muestra los heads actuales
7	alembic stamp head	Marca como aplicada sin ejecutar
