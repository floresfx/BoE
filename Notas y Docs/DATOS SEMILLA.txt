from sqlalchemy.orm import Session

from app.models import Vertical

VERTICALS = [
    {"name": "Clínicas generales", "description": "Clínicas médicas generales"},
    {"name": "Clínicas dentales", "description": "Consultorios odontológicos"},
    {"name": "Panaderías", "description": "Panaderías y pastelerías"},
    {"name": "Restaurantes", "description": "Restaurantes y servicios de comida"},
    {"name": "Fabricantes", "description": "Empresas de manufactura"},
    {"name": "Instituciones educativas", "description": "Colegios, academias, universidades"},
    {"name": "Proyectos por encargo", "description": "Empresas de proyectos y eventos"},
    {"name": "Empresas de limpieza", "description": "Servicios de limpieza"},
    {"name": "Empresas de mantenimiento", "description": "Servicios de mantenimiento"},
    {"name": "Empresas de transporte", "description": "Servicios de transporte y logística"},
]


def seed_verticals(db: Session) -> int:
    """Crea los verticales iniciales."""
    count = 0
    for data in VERTICALS:
        existing = db.query(Vertical).filter(Vertical.name == data["name"]).first()
        if not existing:
            db.add(Vertical(**data))
            count += 1
    db.commit()
    return count

6. seeds/run.py
python
"""Script para ejecutar los datos semilla."""

from app.core.database import SessionLocal
from seeds.permissions import seed_permissions
from seeds.plans import seed_plans
from seeds.profiles import seed_profiles
from seeds.verticals import seed_verticals


def run_seeds() -> None:
    """Ejecuta todos los seeds."""
    db = SessionLocal()
    try:
        print("🌱 Iniciando seeds...")

        v = seed_verticals(db)
        print(f"   ✅ Verticales: {v} creados")

        p = seed_plans(db)
        print(f"   ✅ Planes: {p} creados")

        pr = seed_profiles(db)
        print(f"   ✅ Perfiles: {pr} creados")

        pe = seed_permissions(db)
        print(f"   ✅ Permisos: {pe} creados")

        print("🌱 Seeds completados")
    except Exception as e:
        print(f"❌ Error: {e}")
        db.rollback()
        raise
    finally:
        db.close()


if __name__ == "__main__":
    run_seeds()

from sqlalchemy.orm import Session

from app.core.security import hash_password
from app.models import Role, Tenant, User


def seed_admin(db: Session) -> None:
    """Crea un tenant y admin de prueba."""
    # Verificar si ya existe
    existing = db.query(Tenant).filter(Tenant.tax_id == "J-00000000-0").first()
    if existing:
        print("   ⚠️  Admin ya existe")
        return

    # Crear tenant
    tenant = Tenant(
        name="Tenant de Prueba",
        tax_id="J-00000000-0",
        tax_id_type="RIF",
        admin_email="admin@test.com",
        timezone="America/Caracas",
        country="Venezuela",
        province="Miranda",
        city="Caracas",
        schema_name="tenant_test_admin",
        status="activo",
    )
    db.add(tenant)
    db.flush()

    # Crear rol Admin
    role = Role(
        tenant_id=tenant.id,
        name="Admin",
        description="Administrador",
        is_system=False,
    )
    db.add(role)
    db.flush()

    # Crear usuario Admin
    user = User(
        tenant_id=tenant.id,
        email="admin@test.com",
        password_hash=hash_password("admin123"),
        first_name="Admin",
        last_name="Test",
        phone="04141234567",
        phone_verified=True,
        status="activo",
    )
    db.add(user)
    db.commit()

    print(f"   ✅ Admin creado: admin@test.com / admin123")
