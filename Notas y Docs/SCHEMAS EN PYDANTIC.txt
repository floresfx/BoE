python
from datetime import datetime
from uuid import UUID

from pydantic import BaseModel, ConfigDict


class BaseSchema(BaseModel):
    """Schema base para todos los schemas."""
    model_config = ConfigDict(from_attributes=True)


class TimestampSchema(BaseSchema):
    """Schema con timestamps."""
    created_at: datetime
    updated_at: datetime


class UUIDResponse(BaseSchema):
    """Respuesta con solo UUID."""
    id: UUID


class MessageResponse(BaseSchema):
    """Respuesta con mensaje."""
    message: str


class ErrorResponse(BaseSchema):
    """Respuesta de error."""
    detail: str
2. app/schemas/tenant.py
python
from datetime import datetime
from uuid import UUID

from pydantic import BaseModel, EmailStr, Field

from app.schemas.common import BaseSchema


class TenantBase(BaseModel):
    """Campos base del tenant."""
    name: str = Field(..., min_length=1, max_length=255)
    tax_id: str = Field(..., min_length=1, max_length=50)
    tax_id_type: str = Field(..., min_length=1, max_length=20)
    admin_email: EmailStr
    base_currency: str = Field(default="USD", max_length=10)
    timezone: str = Field(default="America/Caracas", max_length=50)
    country: str = Field(..., min_length=1, max_length=100)
    province: str | None = Field(default=None, max_length=100)
    city: str | None = Field(default=None, max_length=100)


class TenantCreate(TenantBase):
    """Schema para crear un tenant."""
    vertical_id: UUID | None = None
    plan_id: UUID | None = None


class TenantUpdate(BaseModel):
    """Schema para actualizar un tenant."""
    name: str | None = Field(default=None, min_length=1, max_length=255)
    tax_id: str | None = Field(default=None, min_length=1, max_length=50)
    tax_id_type: str | None = Field(default=None, min_length=1, max_length=20)
    admin_email: EmailStr | None = None
    base_currency: str | None = Field(default=None, max_length=10)
    timezone: str | None = Field(default=None, max_length=50)
    country: str | None = Field(default=None, min_length=1, max_length=100)
    province: str | None = Field(default=None, max_length=100)
    city: str | None = Field(default=None, max_length=100)
    vertical_id: UUID | None = None
    plan_id: UUID | None = None
    status: str | None = None


class TenantResponse(TenantBase):
    """Schema de respuesta del tenant."""
    id: UUID
    vertical_id: UUID | None
    plan_id: UUID | None
    status: str
    created_at: datetime
    updated_at: datetime
    model_config = {"from_attributes": True}


class TenantListResponse(BaseSchema):
    """Lista de tenants."""
    items: list[TenantResponse]
    total: int
3. app/schemas/user.py
python
from datetime import datetime
from uuid import UUID

from pydantic import BaseModel, EmailStr, Field

from app.schemas.common import BaseSchema


class UserBase(BaseModel):
    """Campos base del usuario."""
    email: EmailStr
    first_name: str = Field(..., min_length=1, max_length=100)
    last_name: str = Field(..., min_length=1, max_length=100)
    phone: str | None = Field(default=None, max_length=50)


class UserCreate(UserBase):
    """Schema para crear un usuario."""
    password: str = Field(..., min_length=8, max_length=100)
    tenant_id: UUID


class UserUpdate(BaseModel):
    """Schema para actualizar un usuario."""
    email: EmailStr | None = None
    first_name: str | None = Field(default=None, min_length=1, max_length=100)
    last_name: str | None = Field(default=None, min_length=1, max_length=100)
    phone: str | None = Field(default=None, max_length=50)
    status: str | None = None


class UserResponse(UserBase):
    """Schema de respuesta del usuario."""
    id: UUID
    tenant_id: UUID
    phone_verified: bool
    status: str
    last_login_at: datetime | None
    created_at: datetime
    updated_at: datetime
    model_config = {"from_attributes": True}


class UserListResponse(BaseSchema):
    """Lista de usuarios."""
    items: list[UserResponse]
    total: int


class UserLogin(BaseModel):
    """Schema para login."""
    email: EmailStr
    password: str


class UserLoginResponse(BaseSchema):
    """Respuesta del login."""
    access_token: str
    token_type: str = "bearer"
    user: UserResponse
