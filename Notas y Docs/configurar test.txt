from fastapi.testclient import TestClient
from sqlalchemy.orm import Session

from app.models import User


def test_login_exitoso(client: TestClient, user: User):
    """Test: login con credenciales correctas."""
    response = client.post(
        "/api/v1/auth/login",
        json={"email": "test@test.com", "password": "password123"},
    )
    assert response.status_code == 200
    data = response.json()
    assert "access_token" in data
    assert data["token_type"] == "bearer"
    assert data["user"]["email"] == "test@test.com"


def test_login_password_incorrecta(client: TestClient, user: User):
    """Test: login con password incorrecta."""
    response = client.post(
        "/api/v1/auth/login",
        json={"email": "test@test.com", "password": "wrongpassword"},
    )
    assert response.status_code == 401
    assert "Credenciales inválidas" in response.json()["detail"]


def test_login_email_no_existe(client: TestClient):
    """Test: login con email que no existe."""
    response = client.post(
        "/api/v1/auth/login",
        json={"email": "noexiste@test.com", "password": "password123"},
    )
    assert response.status_code == 401


def test_login_usuario_inactivo(client: TestClient, user: User, db: Session):
    """Test: login con usuario inactivo."""
    user.status = "inactivo"
    db.commit()

    response = client.post(
        "/api/v1/auth/login",
        json={"email": "test@test.com", "password": "password123"},
    )
    assert response.status_code == 403


def test_get_me(client: TestClient, auth_headers: dict, user: User):

python
from fastapi.testclient import TestClient

from app.models import User


def test_create_device(client: TestClient, auth_headers: dict):
    """Test: crear un dispositivo seguro."""
    response = client.post(
        "/api/v1/secure-devices",
        headers=auth_headers,
        json={
            "fingerprint": "device-fingerprint-1",
            "ip": "192.168.1.1",
        },
    )
    assert response.status_code == 201


def test_create_device_maximo(client: TestClient, auth_headers: dict):
    """Test: no permitir más de 2 dispositivos."""
    # Crear 2
    for i in range(2):
        client.post(
            "/api/v1/secure-devices",
            headers=auth_headers,
            json={"fingerprint": f"fp-{i}", "ip": "192.168.1.1"},
        )

    # Intentar crear el tercero
    response = client.post(
        "/api/v1/secure-devices",
        headers=auth_headers,
        json={"fingerprint": "fp-3", "ip": "192.168.1.1"},
    )
    assert response.status_code == 400


def test_list_devices(client: TestClient, auth_headers: dict):
    """Test: listar dispositivos."""
    response = client.get("/api/v1/secure-devices", headers=auth_headers)
    assert response.status_code == 200


def test_delete_device(client: TestClient, auth_headers: dict):
    """Test: eliminar un dispositivo."""
    # Crear
    create_response = client.post(
        "/api/v1/secure-devices",
        headers=auth_headers,
        json={"fingerprint": "fp-delete", "ip": "192.168.1.1"},
    )
    device_id = create_response.json()["id"]

    # Eliminar
    response = client.delete(
        f"/api/v1/secure-devices/{device_id}",
        headers=auth_headers,
    )
    assert response.status_code == 204
